```

...
std::string::operator[](i) = SnakeGame::hex_array1[i] ^ original[i];
```
2. 可以在 hex_array1 的內容中可以找到這個陣列共有 43 個 byte，作為 XOR 金鑰。
```
.data:004E3020 hex_array1 db C0h, 19h, 3Ah, FDh, ...
```
3. 程式中有這段被 XOR 的原始密文，這些整數經過 std::string::basic_string() 被轉成原始字串，總共也是 43 bytes。

v16 = 40;     // char

## AIS3 Tiny Server - Reverse
1. 初步分析:
sub_2110: 此函數負責讀取和解析來自客戶端的 HTTP 請求。它包含檢查特定標頭的邏輯，然後呼叫 sub_1E20 來驗證潛在的 flag。
sub_1E20: 此函數包含核心的 flag 檢查機制。它將輸入字串與一個動態產生的目標字串進行比較。
2. 請求處理與 Flag 觸發 (sub_2110):
sub_2110 函數首先讀取 HTTP 請求行（例如 GET / HTTP/1.1）。在解析完這個初始行之後，它進入一個迴圈來讀取後續的 HTTP 標頭。
迴圈 (`do { ... } while (v16[0] != 0x33584F41 || v16[1] != 0x616C462D || v17 != 0x67);`) 正在尋找一個非常特定的標頭。迴圈會一直持續，直到它找到一個以特定位元組序列開頭的行：
`v16[0] == 0x33584F41`: 在小端序中，這是 41 4F 58 33，對應於 ASCII 字串 "AOX3"。
`v16[1] == 0x616C462D`: 在小端序中，這是 2D 46 6C 61，對應於 ASCII 字串 "-Fla"。
`v17 == 0x67`: 這是 ASCII 字元 'g'。
所以，伺服器正在尋找一個以 AOX3-Flag 開頭的標頭行。一旦找到這個標頭，程式碼會提取此標頭的值部分。它會跳過 "AOX3-Flag" 前綴以及任何後續的冒號或空格。產生的字串 (p_s) 然後被傳遞給 sub_1E20 函數進行驗證。
3. Flag 解密邏輯 (sub_1E20)
sub_1E20 函數是實際隱藏和檢查 flag 的地方。將提供的字串（來自 "AOX3-Flag" 標頭）作為輸入 (a1)。 它動態產生一個 45 位元組的目標 flag 字串，並將輸入與之比較。
4. 提取 Flag
要找到 flag，需要重新實作 sub_1E20 中的產生邏輯。
```
def solve_flag():
    # 從 C 程式碼中 sub_1E20 函數獲取的初始資料
    # v8[0] = 1480073267; ... v8[10] = 1246382110;
    v8_dwords_initial = [
        1480073267, 1197221906, 254628393, 920154, 1343445007,
        874076697, 1127428440, 1510228243, 743978009, 54940467,
        1246382110
    ]
    # v9 = 20; (作為第45個位元組)
    v9_last_byte = 20

    # 將 DWORD 轉換為 little-endian 的位元組序列
    original_data_bytes = bytearray()
    for dword in v8_dwords_initial:
        original_data_bytes.extend(dword.to_bytes(4, 'little'))
    
    # 添加來自 v9 的最後一個位元組
    original_data_bytes.append(v9_last_byte) # 總共 45 個位元組

    # 金鑰字串 v7
    key_material = "rikki_l0v3".encode('ascii')
    key_len = len(key_material)

    # C 程式碼中迴圈開始前的初始值
    # v2 = 51;
    operand_v2 = 51
    # v3 = 114;
    operand_v3 = 114

    decrypted_flag_bytes = bytearray(45)

    # 模擬 C 程式碼中的解密迴圈
    # for v1 = 0 to 44:
    #   *((_BYTE *)v8 + v1) = v2 ^ v3;
    #   if ( ++v1 == 45 ) break;
    #   v2 = *((_BYTE *)v8 + v1); // 使用 original_data_bytes[v1]
    #   v3 = v7[v1 % 0xA];
    for i in range(45):
        decrypted_flag_bytes[i] = operand_v2 ^ operand_v3
        
        # 如果不是最後一次迭代，則為下一次迭代準備 operand_v2 和 operand_v3
        if i < 44:
            # v1 在 C 程式碼中此時是 i + 1
            next_index = i + 1
            operand_v2 = original_data_bytes[next_index]
            operand_v3 = key_material[next_index % key_len]

    try:
        flag = decrypted_flag_bytes.decode('ascii')
        return flag
    except UnicodeDecodeError:
        return f"無法解碼為 ASCII: {decrypted_flag_bytes.hex()}"

if __name__ == "__main__":
    flag = solve_flag()
    print(f"解密得到的 Flag: {flag}")
```

import struct

# WebAssembly flagchecker 函數中的常數值
constants = [
    7577352992956835434,   # 0x6935655F67616C66
    7148661717033493303,   # 0x633D3433525F5937
    -7081446828746089091,  # 0x9DCE4BDDC9AE14DD 
    -7479441386887439825,  # 0x9827BCC4DCB12F2F 
    8046961146294847270,   # 0x6FBF7DDE1CFF2876
]

# 用於計算旋轉位數的值
MAGIC = -39934163  # 0xFDA0044D

def rotr64(value, shift):
    """將 64 位值向右旋轉 shift 位 (逆轉 WASM 中的 func8 操作)"""
    # 轉換為無符號整數以進行位元操作
    value &= 0xFFFFFFFFFFFFFFFF
    shift &= 0x3F  # 確保 shift 在 0-63 範圍內
    return ((value >> shift) | (value << (64 - shift))) & 0xFFFFFFFFFFFFFFFF

# 從常數中提取旗標字符
flag_bytes = bytearray()
for i in range(5):
    # 計算旋轉位數 (與 WASM 中相同)
    shift = (MAGIC >> (i * 6)) & 0x3F
    
    # 應用逆向旋轉以獲取原始字節
    original = rotr64(constants[i], shift)
    
    # 轉換為字節並添加到旗標
    original_bytes = struct.pack("<Q", original)
    flag_bytes.extend(original_bytes)

# 將字節轉換為字符串，在第一個空字節處停止
flag = flag_bytes.split(b'\x00')[0].decode('ascii')

print(f"解密的旗標: {flag}")

# 輸出每個塊的具體信息，幫助驗證
print("\n詳細解密過程:")
for i in range(5):
    shift = (MAGIC >> (i * 6)) & 0x3F
    original = rotr64(constants[i], shift)
    chunk_bytes = struct.pack("<Q", original)
    try:
        chunk_str = chunk_bytes.decode('ascii', errors='replace')
        print(f"塊 {i+1}: 移位={shift}, 解密後='{chunk_str}'")
    except:
        print(f"塊 {i+1}: 移位={shift}, 解密後=二進制數據")

# crypto
## SlowECDSA
* 在 chal.c 發現關鍵漏洞：使用 LCG 生成 ECDSA nonce 值 k。如果攻擊者知道了兩個連續的 k 值，就可以恢復出私鑰。接著就讓 AI 幫忙寫個腳本吧!
1. 連續兩次呼叫 get_example 以取得對同一訊息 "example_msg" 的兩個簽名：
第一個簽名 (r1, s1) 使用 nonce k1
第二個簽名 (r2, s2) 使用 nonce k2 = (a * k1 + c) mod n
2. 已知：
s1 = k1^(-1) * (h + r1 * d) mod n
s2 = k2^(-1) * (h + r2 * d) mod n
k2 = (a * k1 + c) mod n
其中 h 是訊息 "example_msg" 的雜湊值，d 是私鑰。
3. 從上述方程中，可以解出`k1 = (h * (r1 - r2) - s2 * c * r1) * inv(s2 * a * r1 - r2 * s1, n) mod n`
4. 一旦獲得了 k1，就可以計算私鑰：`d = (s1 * k1 - h) * inv(r1, n) mod n`
5. 使用恢復的私鑰 d 來簽署訊息 "give_me_flag"：
選擇一個新的隨機 nonce k_flag
計算 r_flag = (k_flag * G).x mod n
計算 s_flag = k_flag^(-1) * (h_flag + r_flag * d) mod n

import hashlib 
from ecdsa import SigningKey, NIST192p

from Crypto.Util.number import inverse, getRandomRange
from pwn import *

order = curve.order

c_lcg = 12345
example_msg_bytes = b"example_msg"

    return int.from_bytes(hashlib.sha1(msg_bytes).digest(), 'big') % current_order # <--- 確保此函數已正確實現

# 範例：如何與伺服器互動 (使用 pwntools)

def get_sig_from_server():
    io.sendlineafter(b"Enter option: ", b"get_example")
    io.recvuntil(b"msg: example_msg\n")
    r_line = io.recvline().decode().strip()
    s_line = io.recvline().decode().strip()
    r_val = int(r_line.split(": ")[1], 16)
    s_val = int(s_line.split(": ")[1], 16)
    return r_val, s_val

print("[+] Getting second signature from server...")
r2, s2 = get_sig_from_server()
print(f"    r2: {hex(r2)}")
print(f"    s2: {hex(s2)}")

# 2. 計算 h_example
h_example = calculate_h(example_msg_bytes, order)
print(f"[+] h_example: {h_example}")

# 3. 計算 k1
try:
    term1_num = (h_example * (r1 - r2)) % order
    term2_num = (s2 * c_lcg * r1) % order
    numerator = (term1_num - term2_num + order) % order

    term1_den = (s2 * a_lcg * r1) % order
    term2_den = (r2 * s1) % order
    denominator_val = (term1_den - term2_den + order) % order

    if denominator_val == 0:
        print("[-] Error: Denominator for k1 calculation is zero.")
        io.close()
        exit(1)

    inv_denominator = inverse(denominator_val, order)
    k1 = (numerator * inv_denominator) % order
    print(f"[+] Recovered k1: {k1}")

    # 4. 計算私鑰 sk (d)
    if r1 == 0:
        print("[-] Error: r1 is zero, cannot compute inverse for sk.")
        io.close()
        exit(1)
        
    sk_numerator = (s1 * k1 - h_example + order) % order
    inv_r1 = inverse(r1, order)
    sk_priv_multiplier = (sk_numerator * inv_r1) % order
    print(f"[+] Recovered private key (d): {sk_priv_multiplier}")

    # 5. 簽署 "give_me_flag"
    target_msg_bytes = b"give_me_flag"
    h_flag = calculate_h(target_msg_bytes, order)

    # 先用隨機 k
    k_flag = getRandomRange(1, order - 1)

    R_flag_point = k_flag * curve.generator
    r_flag = R_flag_point.x() % order

    if r_flag == 0:
        print("[-] Error: r_flag is zero, try a different k_flag or re-run.")
        io.close()
        exit(1)

    s_flag_inv_k = inverse(k_flag, order)
    s_flag = (s_flag_inv_k * (h_flag + r_flag * sk_priv_multiplier)) % order

    if s_flag == 0:
        print("[-] Error: s_flag is zero, try a different k_flag or re-run.")
        io.close()
        exit(1)
        
    print(f"[+] Signature for 'give_me_flag':")
    print(f"    msg: {target_msg_bytes.decode()}")
    print(f"    r: {hex(r_flag)}")
    print(f"    s: {hex(s_flag)}")

    # 6. 將簽名提交給伺服器的 "verify" 選項
    def submit_signature(msg_str, r_hex_str, s_hex_str):
        io.sendlineafter(b"Enter option: ", b"verify")
        io.sendlineafter(b"Enter message: ", msg_str.encode())
        io.sendlineafter(b"Enter r (hex): ", r_hex_str.encode()) # r_hex_str 
        io.sendlineafter(b"Enter s (hex): ", s_hex_str.encode()) # s_hex_str 
        response = io.recvall(timeout=2) # 讀取伺服器回應
        print(response.decode())

    print("[+] Submitting signature for 'give_me_flag' to server...")

## Stream
* 發現這題的關鍵在於 Python 的 random 模組使用的是 Mersenne Twister (MT19937) 虛擬隨機數生成器，這是一個確定性的偽隨機數生成器。如果能獲取連續的 624 個 32 位元輸出，就能夠完全預測後續的輸出。
* 由於 getrandbits(256) 實際上是通過連續生成 8 個 32 位元隨機數並拼接而成的（256/32 = 8），所以從 80 次迴圈中可以獲得 80 * 8 = 640 個 32 位元隨機數，足夠破解 MT19937 的內部狀態。

solve.py:
```
from hashlib import sha512
import math
from randcrack import RandCrack

def solve():
    # 1. 解析output.txt
    with open("output.txt", "r") as f:
        output_lines_hex = [line.strip() for line in f.readlines()]
    
    output_values_int = [int(h, 16) for h in output_lines_hex]

    # 2. 預先計算所有可能的 SHA-512 哈希值
    H_lookup = [int.from_bytes(sha512(bytes([i])).digest(), 'big') for i in range(256)]

    # 3. 和 4. 恢復 r_i 值和它們的 32 位元 MT 組件
    mt_outputs_32bit = []
    
    print("Recovering r_i values...")
    for i in range(80): # First 80 lines
        output_val = output_values_int[i]
        found_match_for_line = False
        for h_candidate_val in H_lookup:
            potential_square = output_val ^ h_candidate_val
            if potential_square < 0: continue

            # 使用 math.isqrt 計算整數平方根
            r_candidate = math.isqrt(potential_square)
            if r_candidate * r_candidate == potential_square:
                if 0 <= r_candidate < (1 << 256): #檢查 r_candidate 是否為 256 位元數
                    # r_i
                    for k_idx in range(8): # 256 bits = 8 * 32 bits
                        u_k = (r_candidate >> (32 * k_idx)) & 0xFFFFFFFF
                        mt_outputs_32bit.append(u_k)
                    found_match_for_line = True
                    break # 找到匹配項，結束搜索
        
        if not found_match_for_line:
            print(f"Error: Could not find match for output line {i}: {output_lines_hex[i]}")
            return "Failed to recover r_i for all lines."
    
    if len(mt_outputs_32bit) < 640: # Should be 80 * 8 = 640
        print(f"Error: Not enough MT outputs recovered ({len(mt_outputs_32bit)}), expected 640.")
        return "Failed to recover sufficient MT outputs."
    print(f"Recovered {len(mt_outputs_32bit)} 32-bit MT outputs.")

    # 5. 破解 Mersenne Twister
    print("Cracking Mersenne Twister state...")
    cracker = RandCrack()
    for j in range(624): # 輸入前 624 個 32 位元輸出
        cracker.submit(mt_outputs_32bit[j])
    print("MT state seeded.")

    # 6. 預測 r_flag
    # 80 個循環使用了 u_0 到 u_639
    # 已經將 u_0 到 u_623 輸入到破解器
    # 需要預測 u_640 到 u_647 來獲得 r_flag
    # 首先，將破解器的狀態推進到已知的 u_624...u_639 值之後
    # 這些是 640 - 624 = 16 個值
    print("Advancing cracker state...")
    for _ in range(16): # This loop consumes predictions for u_624 through u_639
        cracker.predict_getrandbits(32)

    # 現在預測 r_flag 的 8 個組件（u_640 到 u_647）
    print("Predicting r_flag components...")
    r_flag_components_pred = []
    for _ in range(8):
        pred_u_k = cracker.predict_getrandbits(32)
        r_flag_components_pred.append(pred_u_k)

    r_flag_predicted_int = 0
    for k_idx in range(8): # Assemble r_flag from its 8 components
        r_flag_predicted_int |= (r_flag_components_pred[k_idx] << (32 * k_idx))
    print(f"Predicted r_flag: {hex(r_flag_predicted_int)}")

    # 7. 計算 Flag
    o_flag_val = output_values_int[-1] # 輸出的最後一行含有 flag

    int_flag = o_flag_val ^ (r_flag_predicted_int**2)

    if int_flag == 0:
        flag_bytes = b''
    else:
        byte_len = (int_flag.bit_length() + 7) // 8
        flag_bytes = int_flag.to_bytes(byte_len, 'big')
    
    try:
        flag_str = flag_bytes.decode('utf-8')
        print(f"Successfully decoded flag: {flag_str}")
        return flag_str
    except UnicodeDecodeError:
        print(f"Could not decode flag bytes: {flag_bytes.hex()}")
        return f"Could not decode flag bytes: {flag_bytes.hex()}"

if __name__ == '__main__':
    result = solve()
```

1. 需要從提示中恢復出 LCG 的參數 a 和 b。現在知道：
h0 = (a * seed + b) % M
h1 = (a * h0 + b) % M
h2 = (a * h1 + b) % M

h2 - h1 = a * (h1 - h0) % M`求解
從第二個等式可以解出`a = ((h2 - h1) * inverse(h1 - h0, M)) % M`

2. q 是通過對 p 應用 genPrime() 函數得到的。而 genPrime() 函數執行：

這意味著 q 是通過對 p 多次應用 LCG 得到的。即`q = rng^j(p) = (A_j * p + B_j) % M`
3. 由於 n = p * q 且 q = (A_j * p + B_j) % M，可以推導出：`n/p ≡ A_j * p + B_j (mod M)`
兩邊同乘以 p（假設 p 不是 M 的倍數）：`n ≡ A_j * p² + B_j * p (mod M)`
這可以寫成二次同餘方程：`A_j * p² + B_j * p - n ≡ 0 (mod M)`
4. 解二次同餘方程求 p
對於每個可能的 j 值計算 A_j 和 B_j
求解二次同餘方程 A_j * x² + B_j * x - n ≡ 0 (mod M)
5. 一旦找到正確的 p 和 q：
計算 `phi = (p - 1) * (q - 1)`
計算 `d = inverse(e, phi)`
解密 `m = c^d mod n`
將 m 轉換為字節得到 flag
